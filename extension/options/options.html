const form = document.getElementById("ogc-options");

document.querySelectorAll("[data-i18n]").forEach((el) => {
  const msg = browser.i18n.getMessage(el.dataset.i18n);
  if (msg) el.textContent = msg;
});

(async () => {
  const { settings = {} } = await browser.storage.local.get("settings");
  for (const [k, v] of Object.entries(settings)) {
    const el = form.elements[k];
    if (!el) continue;
    if (el.type === "checkbox") el.checked = !!v;
    else el.value = v;
  }
  document.getElementById("vocab").textContent = JSON.stringify(OGC_VOCABULARY, null, 2);
})();

form.addEventListener("submit", async (e) => {
  e.preventDefault();
  const fd = new FormData(form);
  const settings = {
    enabled: form.elements.enabled.checked,
    trails: form.elements.trails.checked,
    tooltips: form.elements.tooltips.checked,
    button: Number(fd.get("button")),
    sensitivity: Number(fd.get("sensitivity"))
  };
  await browser.storage.local.set({ settings });
});
